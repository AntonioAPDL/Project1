\section*{1. Executive diagnosis}
\begin{tcolorbox}[colback=Soft,colframe=Accent,boxrule=0.6pt,arc=2mm]
\textbf{Main reasons for rejection-with-invitation-to-resubmit:}
\begin{enumerate}[leftmargin=1.2em,itemsep=3pt,topsep=2pt]
\item \textbf{Insufficient validation for a forecasting paper}: unclear in-sample vs out-of-sample; no rigorous cross-validation; evaluation limited to a short period/event; missing proper forecast verification summaries.
\item \textbf{No compelling benchmarking / ablation}: few (or no) competing alternatives; unclear contribution of hierarchical components; no evidence extended asymmetric Laplace (exAL) meaningfully helps vs standard AL.
\item \textbf{Missing forecast distribution scoring and reporting}: if CRPS is used in estimation/selection, reviewers expect CRPS (and quantile score) \emph{reported} for forecasting; diagnostics by regime/event type.
\item \textbf{Messaging/structure mismatch}: abstract claims ``fast'' but no runtime; introduction mixes meteorological vs hydrological uncertainty; too much math detail not used (PIT section) and over-detailed PFS/quantile-crossing treatment.
\item \textbf{Usability and reproducibility}: code availability requested; end-user interpretability requires ``bang-for-buck'' narrative.
\end{enumerate}
\end{tcolorbox}

% ============================================================
\section*{2. Highest-priority deliverables (recommended order)}
\begin{enumerate}[leftmargin=1.2em,itemsep=4pt,topsep=2pt]
\item \textbf{Define the evaluation protocol}: strict out-of-sample forecasting with time-split CV / rolling-origin evaluation; explicitly state training window, forecast horizon(s), and update strategy.
\item \textbf{Benchmarking}: compare against (i) simpler statistical baselines, (ii) the original physical forecast ensembles/products, and (iii) AL vs exAL.
\item \textbf{Ablation study}: remove/disable hierarchical components (trend AR, retrospective AR, state AR, horizon-weighting) to quantify marginal benefit.
\item \textbf{Forecast scoring + reporting}: CRPS, quantile score (pinball/check), calibration diagnostics; stratify by regime (low/moderate/high flow) and by lead time.
\item \textbf{Computational claims}: report runtime, scaling (n, horizons, #quantiles), memory, and feasibility of re-estimation/updating over months/years.
\item \textbf{Restructure paper}: introduction uncertainty taxonomy; align Section 2 models with Section 3 results; reduce unused math; clarify in-sample vs forecasting sections/titles.
\item \textbf{Release code}: minimal reproducible pipeline; link in paper + repository artifact DOI if possible.
\end{enumerate}
